cmake_minimum_required(VERSION 3.16)
project(passport_ui_preview C CXX)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(preview preview.c ../../main/font_brand_22.c ../../components/passport_core/passport_menu.c ../../components/passport_core/passport_model.c)
target_include_directories(preview PRIVATE ../../main ../../components/passport_core/include)
target_link_libraries(preview PRIVATE lvgl m)
target_link_options(preview PRIVATE -fsanitize=address)
